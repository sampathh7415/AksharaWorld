```typescript
// /src/lib/automation/order-processor.ts (AUTO-GENERATED)
export async function processOrderToDelivery(webhookPayload) {
  try {
    // 1. Verify webhook signature
    const verified = await verifyRazorpaySignature(webhookPayload);
    if (!verified) throw new Error('Invalid signature');

    // 2. Extract order details
    const order = extractOrderData(webhookPayload);
    
    // 3. Log to Sheets
    await logToSheets('Orders', {
      timestamp: new Date(),
      ...order,
      status: 'PAID'
    });

    // 4. Send Telegram alert
    await sendTelegramAlert(`New order: ${order.name} - ₹${order.amount}`);

    // 5. Fetch resume
    const resume = await fetchCustomerResume(order.email);

    // 6. Analyze with Ollama
    const analysis = await analyzeResumeWithOllama(resume);

    // 7. Generate PDF
    const pdf = await generateAnalysisPDF(analysis, order);

    // 8. Upload to Drive
    const driveLink = await uploadToDrive(pdf, order.id);

    // 9. Send email
    await sendDeliveryEmail(order.email, {
      pdf,
      driveLink,
      analysis
    });

    // 10. Update Sheets
    await updateSheets('Orders', order.id, {
      status: 'DELIVERED',
      delivered_at: new Date(),
      drive_link: driveLink
    });

    // 11. Send completion alert
    await sendTelegramAlert(`Order #${order.id} DELIVERED`);

    // 12. Schedule follow-ups
    await scheduleFollowUps(order);

    return { success: true, orderId: order.id };
  } catch (error) {
    await logError(error);
    await sendTelegramAlert(`❌ Order processing failed: ${error.message}`);
    throw error;
  }
}
```
